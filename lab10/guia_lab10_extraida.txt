capturar.sh
#!/usr/bin/env bash
# capturar.sh — "La prueba del espía": inicia sesión por TELNET o por SSH mientras tcpdump graba, y muestra qué
# se puede leer de la captura. Uso (en el servidor, como root):  bash capturar.sh telnet   |   bash capturar.sh ssh
set -uo pipefail
cd "$(dirname "$0")"
PROTO="${1:?Uso: bash capturar.sh telnet|ssh}"
USUARIO=secretaria; CLAVE=Circular2026; SECRETO="notas-confidenciales-5to-B"
[[ $EUID -eq 0 ]] || { echo "Ejecuta como root"; exit 1; }
command -v expect >/dev/null && command -v tcpdump >/dev/null || apt-get install -y -qq expect tcpdump >/dev/null
# Usuaria de prueba (solo para el laboratorio)
id "$USUARIO" &>/dev/null || useradd -m -s /bin/bash "$USUARIO"
echo "$USUARIO:$CLAVE" | chpasswd
case $PROTO in
  telnet) PUERTO=23; CMD="telnet 127.0.0.1"; PIDE="login:" ;;
  ssh)    PUERTO=22; CMD="ssh -o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -o PubkeyAuthentication=no $USUARIO@127.0.0.1"; PIDE="password:" ;;
  *) echo "Protocolo desconocido: $PROTO"; exit 1 ;;
esac
ss -tln | grep -q ":$PUERTO " || { echo "Nada escucha en el puerto $PUERTO (¿activaste el servicio?)"; exit 1; }
mkdir -p resultados
PCAP=resultados/$PROTO.pcap
timeout 15 tcpdump -i lo -nn -U -w "$PCAP" "tcp port $PUERTO" 2>/dev/null &
sleep 1
expect <<EOF >/dev/null 2>&1
set timeout 10
spawn $CMD
expect "$PIDE"
if {"$PROTO" == "telnet"} { send "$USUARIO\r"; expect "Password:" }
send "$CLAVE\r"
expect "\\\$ "
send "echo $SECRETO\r"
expect "\\\$ "
send "exit\r"
expect eof
EOF
sleep 2; pkill -f "tcpdump -i lo -nn -U -w $PCAP" 2>/dev/null; sleep 1
TEXTO=$(tcpdump -r "$PCAP" -A 2>/dev/null)
echo "Captura de $PROTO: $(tcpdump -r "$PCAP" 2>/dev/null | wc -l) paquetes → $PCAP (ábrelo en Wireshark)"
for dato in "$USUARIO" "$CLAVE" "$SECRETO"; do
  if grep -qa -- "$dato" <<<"$TEXTO"; then echo "  👀 se lee en claro: $dato"; else echo "  🔒 no aparece:     $dato"; fi
done
grep -aoE 'SSH-2.0-[A-Za-z0-9_.-]+' <<<"$TEXTO" | sort -u | sed 's/^/  versión anunciada: /'
{ echo "$PROTO $(date -Iseconds)"; for d in "$USUARIO" "$CLAVE" "$SECRETO"; do grep -qa -- "$d" <<<"$TEXTO" && echo "claro $d" || echo "oculto $d"; done; } > resultados/$PROTO.txt
telnet_laboratorio.sh
#!/usr/bin/env bash
# telnet_laboratorio.sh — Activa o desactiva TELNET SOLO para la demostración. Nunca lo dejes activo.
# Uso:  bash telnet_laboratorio.sh activar   |   bash telnet_laboratorio.sh desactivar
set -euo pipefail
[[ $EUID -eq 0 ]] || { echo "Ejecuta como root"; exit 1; }
case "${1:-}" in
  activar)
    dpkg -s telnetd &>/dev/null || { apt-get update -qq; apt-get install -y -qq telnetd openbsd-inetd inetutils-telnet >/dev/null; }
    update-inetd --enable telnet            # Ubuntu lo instala DESACTIVADO (#<off># en /etc/inetd.conf)
    systemctl restart openbsd-inetd
    echo "⚠️  TELNET activo en el puerto 23. Desactívalo al terminar: bash $0 desactivar" ;;
  desactivar)
    update-inetd --disable telnet 2>/dev/null || true
    systemctl disable --now openbsd-inetd 2>/dev/null || true
    apt-get purge -y -qq telnetd >/dev/null 2>&1 || true
    ss -tln | grep -q ':23 ' && echo "❌ El puerto 23 sigue abierto" || echo "✅ TELNET desactivado y desinstalado" ;;
  *) echo "Uso: bash $0 activar|desactivar"; exit 1 ;;
esac

#!/usr/bin/env bash
# preparar_ssh.sh — Identidad propia del servidor y acceso por clave ed25519 para el administrador.
# Uso (en el servidor, como root):  bash preparar_ssh.sh <usuario_admin>     (el usuario del Lab 2, p. ej. ana)
set -euo pipefail
cd "$(dirname "$0")"
ADMIN="${1:?Uso: bash preparar_ssh.sh <usuario_admin>}"
[[ $EUID -eq 0 ]] || { echo "Ejecuta como root"; exit 1; }
id "$ADMIN" &>/dev/null || { echo "No existe $ADMIN (Lab 2)"; exit 1; }
# 1. Claves de HOST propias: una imagen o VM clonada trae las mismas claves en todos los equipos (misma "cédula")
if grep -q 'buildkitsandbox\|@localhost' /etc/ssh/ssh_host_ed25519_key.pub 2>/dev/null || [[ ! -f resultados/.claves_host_regeneradas ]]; then
  # TODO: borrar las claves de host actuales (/etc/ssh/ssh_host_*) y generar nuevas (pista: ssh-keygen -A)
  systemctl restart ssh
  mkdir -p resultados && date -Iseconds > resultados/.claves_host_regeneradas
fi
echo "Huella del servidor (compárala en la primera conexión):"
ssh-keygen -lf /etc/ssh/ssh_host_ed25519_key.pub
# 2. Par de claves del CLIENTE (aquí el cliente es root del servidor; en la vida real, tu PC)
# TODO: si no existe ~/.ssh/id_ed25519, crear un par ed25519 (-t), con 64 rondas de protección (-a 64),
#       comentario "admin-horizonte-AAAA" (-C) y sin frase (-N "") — en tu PC real, ¡usa una frase!
# 3. Autorizar la clave pública en la cuenta del administrador (lo que hace ssh-copy-id)
install -d -o "$ADMIN" -g "$ADMIN" -m 700 "/home/$ADMIN/.ssh"
touch "/home/$ADMIN/.ssh/authorized_keys"
grep -qF "$(cut -d' ' -f2 ~/.ssh/id_ed25519.pub)" "/home/$ADMIN/.ssh/authorized_keys" || cat ~/.ssh/id_ed25519.pub >> "/home/$ADMIN/.ssh/authorized_keys"
chown "$ADMIN:$ADMIN" "/home/$ADMIN/.ssh/authorized_keys"
# TODO: permisos correctos de authorized_keys (¿qué hace sshd si otros pueden escribirlo?)
# 4. known_hosts con la huella REAL (sin aceptar a ciegas) y un alias en ~/.ssh/config
ssh-keygen -R 127.0.0.1 >/dev/null 2>&1 || true
ssh-keyscan -t ed25519 127.0.0.1 2>/dev/null >> ~/.ssh/known_hosts
if ! grep -q '^Host horizonte$' ~/.ssh/config 2>/dev/null; then
  cat >> ~/.ssh/config <<EOF
Host horizonte
    HostName 127.0.0.1
    User $ADMIN
    # TODO: IdentityFile con tu clave ed25519 e IdentitiesOnly yes
EOF
fi
chmod 600 ~/.ssh/config
ssh -o BatchMode=yes horizonte 'echo "Conectado como $(whoami) a $(hostname) con clave ed25519"'

verificar_lab10.sh
#!/usr/bin/env bash
# verificar_lab10.sh — Comprueba el Laboratorio 10 (TELNET vs SSH). Ejecutar en el servidor como root:
#   bash verificar_lab10.sh <usuario_admin>
cd "$(dirname "$0")"
ADMIN="${1:-ana}"
ok=0; total=0
revisar() {
  local desc="$1"; shift; total=$((total + 1))
  if "$@" &>/dev/null; then echo "✅ $desc"; ok=$((ok + 1)); else echo "❌ $desc"; fi
}
telnet_en_claro() { grep -q '^claro Circular2026' resultados/telnet.txt; }
ssh_oculto()      { grep -q '^oculto Circular2026' resultados/ssh.txt && grep -q '^oculto notas-confidenciales' resultados/ssh.txt; }
host_propia()     { ! grep -q 'buildkitsandbox' /etc/ssh/ssh_host_ed25519_key.pub && [[ -f resultados/.claves_host_regeneradas ]]; }
known_ok()        { ssh-keygen -F 127.0.0.1 -f ~/.ssh/known_hosts | grep -q "$(awk '{print $2}' /etc/ssh/ssh_host_ed25519_key.pub)"; }
clave_cliente()   { ssh-keygen -lf ~/.ssh/id_ed25519.pub | grep -q '(ED25519)'; }
autorizada()      { grep -qF "$(awk '{print $2}' ~/.ssh/id_ed25519.pub)" "/home/$ADMIN/.ssh/authorized_keys"; }
permisos()        { [[ $(stat -c %a "/home/$ADMIN/.ssh") == 700 && $(stat -c %a "/home/$ADMIN/.ssh/authorized_keys") == 600 ]]; }
entra_con_clave() { [[ "$(ssh -o BatchMode=yes horizonte whoami)" == "$ADMIN" ]]; }
kex_pq()          { ssh -v -o BatchMode=yes horizonte true 2>&1 | grep -q 'kex: algorithm: sntrup761x25519\|kex: algorithm: mlkem768x25519'; }
publicado()       { [[ -f /srv/horizonte/www/comunicado.html && $(stat -c %U:%G /srv/horizonte/www/comunicado.html) == "$ADMIN:webadmin" ]]; }
telnet_apagado()  { ! ss -tln | grep -q ':23 '; }
revisar "Captura TELNET: la contraseña se lee en claro"         telnet_en_claro
revisar "Captura SSH: contraseña y datos NO se leen"            ssh_oculto
revisar "Claves de host propias (no las de la imagen)"          host_propia
revisar "known_hosts con la huella real del servidor"           known_ok
revisar "Clave del cliente ed25519"                             clave_cliente
revisar "Clave pública autorizada en la cuenta de $ADMIN"       autorizada
revisar "Permisos de ~/.ssh (700) y authorized_keys (600)"      permisos
revisar "Acceso con clave y alias 'horizonte' (sin contraseña)"  entra_con_clave
revisar "Intercambio de claves híbrido post-cuántico"           kex_pq
revisar "comunicado.html publicado por scp (dueño $ADMIN, grupo webadmin)" publicado
revisar "TELNET desactivado al terminar (puerto 23 cerrado)"    telnet_apagado
echo "Resultado: $ok de $total comprobaciones"
[[ $ok -eq $total ]]
