SSH de nivel profesional: endurecimiento, SFTP enjaulado, fail2ban y túneles
1. Objetivo
Endurecer OpenSSH (sin root, solo claves, grupos permitidos, criptografía moderna y registro de huellas), crear un acceso SFTP enjaulado para el equipo de comunicación, bloquear la fuerza bruta con fail2ban y usar un túnel SSH hacia un servicio interno, hasta pasar un verificador de 15 comprobaciones.
2. Contexto del problema
El servidor del colegio saldrá a internet con su dominio público. El registro de un servidor de prueba mostró miles de intentos de acceso en un día. Además, la encargada de comunicación necesita subir las fotos de la feria de ciencias sin que nadie le dé la contraseña de administrador.

endurecer_ssh.sh
#!/usr/bin/env bash
# endurecer_ssh.sh — SSH endurecido, SFTP enjaulado para comunicación y fail2ban.
# Uso (en el servidor, como root, DESPUÉS del Lab 10):  bash endurecer_ssh.sh <usuario_admin>
# ⚠️  Regla de oro: deja OTRA sesión SSH abierta mientras cambias sshd. Si te equivocas, la usas para arreglarlo.
set -euo pipefail
cd "$(dirname "$0")"
ADMIN="${1:?Uso: bash endurecer_ssh.sh <usuario_admin>}"
[[ $EUID -eq 0 ]] || { echo "Ejecuta como root"; exit 1; }
# 0. Salvavidas: el administrador DEBE tener clave autorizada, o quedará fuera al apagar las contraseñas
[[ -s /home/$ADMIN/.ssh/authorized_keys ]] || { echo "❌ $ADMIN no tiene authorized_keys: completa el Lab 10 primero"; exit 1; }
# 1. Grupos que pueden usar SSH
getent group sshadmins >/dev/null || groupadd sshadmins
getent group sftp-web  >/dev/null || groupadd sftp-web
usermod -aG sshadmins "$ADMIN"
# 2. Cuenta del equipo de comunicación: solo SFTP para publicar en www (grupo webadmin del Lab 2)
id comunicacion &>/dev/null || useradd -m -s /usr/sbin/nologin -G sftp-web,webadmin comunicacion
[[ -f /root/.ssh/comunicacion_ed25519 ]] || ssh-keygen -q -t ed25519 -C "comunicacion-horizonte" -f /root/.ssh/comunicacion_ed25519 -N ""
install -d -o comunicacion -g comunicacion -m 700 /home/comunicacion/.ssh
install -o comunicacion -g comunicacion -m 600 /root/.ssh/comunicacion_ed25519.pub /home/comunicacion/.ssh/authorized_keys
# 3. La jaula: sshd exige que la carpeta del chroot y todas sus superiores sean de root y NO escribibles por el grupo
# TODO: si la jaula falla con 'bad ownership or modes for chroot directory' (journalctl -u ssh), corrige aquí
#       el dueño y los permisos de /srv/horizonte (NO los de www)
ls -ld /srv/horizonte/www
# 4. Configuración de sshd: validar ANTES de reiniciar
install -m 644 ssh/issue.net /etc/issue.net
install -m 644 ssh/10-horizonte.conf /etc/ssh/sshd_config.d/10-horizonte.conf
sshd -t
systemctl restart ssh
# 5. fail2ban con nftables leyendo el journal
dpkg -s fail2ban &>/dev/null || { apt-get update -qq; apt-get install -y -qq fail2ban nftables python3-systemd >/dev/null; }
install -m 644 fail2ban/horizonte-ssh.local /etc/fail2ban/jail.d/horizonte-ssh.local
systemctl enable fail2ban >/dev/null 2>&1
systemctl restart fail2ban
sleep 2
fail2ban-client status sshd | sed -n '1,4p'
echo "✅ SSH endurecido. Prueba en OTRA terminal: ssh horizonte   y   sftp -i ~/.ssh/comunicacion_ed25519 comunicacion@127.0.0.1"
verificar_lab11.sh
#!/usr/bin/env bash
# verificar_lab11.sh — Comprueba el SSH endurecido, el SFTP enjaulado, fail2ban y los túneles.
# Uso (en el servidor, como root):  bash verificar_lab11.sh <usuario_admin>
cd "$(dirname "$0")"
ADMIN="${1:-ana}"
ok=0; total=0
revisar() {
  local desc="$1"; shift; total=$((total + 1))
  if "$@" &>/dev/null; then echo "✅ $desc"; ok=$((ok + 1)); else echo "❌ $desc"; fi
}
T() { sshd -T | grep -i "^$1 " | cut -d' ' -f2-; }
SFTP=(sftp -b - -i /root/.ssh/comunicacion_ed25519 -o BatchMode=yes -o StrictHostKeyChecking=no comunicacion@127.0.0.1)
root_rechazado() { ssh -o BatchMode=yes -o StrictHostKeyChecking=no root@127.0.0.1 true 2>&1 | grep -q 'Permission denied'; }
admin_entra()    { [[ "$(ssh -o BatchMode=yes horizonte whoami)" == "$ADMIN" ]]; }
admin_grupo()    { id -nG "$ADMIN" | grep -qw sshadmins && T allowgroups | grep -q sshadmins; }
sftp_publica()   { rm -f /srv/horizonte/www/prueba-sftp.txt; echo hola > /root/prueba-sftp.txt
                   printf 'put /root/prueba-sftp.txt prueba-sftp.txt\nchmod 664 prueba-sftp.txt\n' | "${SFTP[@]}" >/dev/null 2>&1
                   [[ $(stat -c '%U:%G %a' /srv/horizonte/www/prueba-sftp.txt) == "comunicacion:webadmin 664" ]]; }
sftp_enjaulado() { printf 'cd /etc\n' | "${SFTP[@]}" 2>&1 | grep -qi 'No such file'; }
sin_terminal()   { ssh -i /root/.ssh/comunicacion_ed25519 -o BatchMode=yes comunicacion@127.0.0.1 id 2>&1 | grep -q 'sftp connections only'; }
kex_pq()         { T kexalgorithms | grep -q '^sntrup761x25519'; }
huella_log()     { journalctl -u ssh --no-pager | grep -q "Accepted publickey for $ADMIN .* ED25519 SHA256:"; }
banner_ascii()   { [[ -s /etc/issue.net ]] && ! LC_ALL=C grep -q '[^[:print:][:space:]]' /etc/issue.net; }
f2b_ve_ssh()     { fail2ban-client status sshd | grep -q '_SYSTEMD_UNIT=ssh.service'; }
f2b_bloquea()    { fail2ban-client set sshd banip 203.0.113.66 >/dev/null; local r=1
                   nft list set inet f2b-table addr-set-sshd 2>/dev/null | grep -q 203.0.113.66 && r=0
                   fail2ban-client set sshd unbanip 203.0.113.66 >/dev/null; return $r; }
tunel_local()    { ssh -o BatchMode=yes -f -N -L 9082:127.0.0.1:8081 horizonte && sleep 1
                   curl -s http://127.0.0.1:9082/eco | grep -q '"metodo"'; local r=$?; pkill -f 'L 9082:127.0.0.1:8081'; return $r; }
revisar "sshd -t sin errores"                                   sshd -t
revisar "root no puede entrar por SSH"                           root_rechazado
revisar "Contraseñas desactivadas (solo claves)"                 test "$(T passwordauthentication)" = no
revisar "$ADMIN entra con su clave (alias horizonte)"             admin_entra
revisar "Solo los grupos sshadmins y sftp-web (y $ADMIN en sshadmins)" admin_grupo
revisar "comunicacion publica por SFTP (archivo 664, grupo webadmin)" sftp_publica
revisar "comunicacion está enjaulado (no ve /etc)"               sftp_enjaulado
revisar "comunicacion no obtiene terminal (solo SFTP)"           sin_terminal
revisar "Intercambio de claves post-cuántico primero"            kex_pq
revisar "El registro guarda la huella de la clave usada"         huella_log
revisar "Aviso legal en ASCII (/etc/issue.net)"                  banner_ascii
revisar "fail2ban activo"                                        systemctl is-active fail2ban
revisar "fail2ban lee el journal de ssh.service"                 f2b_ve_ssh
revisar "fail2ban bloquea en nftables"                           f2b_bloquea
revisar "Túnel local: API interna 8081 alcanzable por SSH"       tunel_local
echo "Resultado: $ok de $total comprobaciones"
[[ $ok -eq $total ]]
